# qwen3.6-27b box queue: one `name: command` per line, run in order from the repo root by queue.sh.
# GPU jobs bring the stack up first (a no-op when healthy), so the retry after a vLLM crash restarts it.
stack: bash scripts/remote/stack.sh up qwen3.6-27b
host-compare: bash scripts/remote/stack.sh up qwen3.6-27b && uv run python scripts/host_compare.py run qwen3.6-27b
collect-calib: bash scripts/remote/stack.sh up qwen3.6-27b && uv run python scripts/collect.py --profile qwen3.6-27b --split calib --concurrency 4
collect-eval: bash scripts/remote/stack.sh up qwen3.6-27b && uv run python scripts/collect.py --profile qwen3.6-27b --split eval --concurrency 4
collect-index-dev: bash scripts/remote/stack.sh up qwen3.6-27b && uv run python scripts/collect.py --profile qwen3.6-27b --split index-dev --concurrency 4
