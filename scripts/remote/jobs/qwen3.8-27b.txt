stack: bash scripts/remote/stack.sh up qwen3.8-27b

jevbench-after: bash scripts/remote/stack.sh up qwen3.8-27b && uv run python scripts/bench.py --profile qwen3.8-27b --jevbench-only

latency-after: bash scripts/remote/stack.sh up qwen3.8-27b && uv run python scripts/latency.py --profile qwen3.8-27b
image-sharing: bash scripts/remote/stack.sh up qwen3.8-27b && uv run python scripts/image_share_probe.py --profile qwen3.8-27b
vision-latency: bash scripts/remote/stack.sh up qwen3.8-27b && uv run python scripts/vision_run.py --profile qwen3.8-27b --gateway http://127.0.0.1:9000 --sample 300 --workers 1 --run-dir .cache/vision-runs/qwen3.8-27b/latency-$(cat .cache/remote/run_tag)

vision-stock: bash scripts/vision_baseline.sh qwen3.8-27b
